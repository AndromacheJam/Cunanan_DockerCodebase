FROM devilbox/php-fpm:8.2-work

# This base image already ships with all the PHP extensions this project
# needs (pdo, pdo_mysql, mysqli, gd, zip, curl), so no compilation step
# is required — we only need to drop in our custom php.ini.

USER root

COPY conf.d/php.ini /usr/local/etc/php/conf.d/custom.ini

WORKDIR /var/www

EXPOSE 9000

CMD ["php-fpm"]
